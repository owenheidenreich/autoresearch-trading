class PositionStateGenerator(nn.Module):
    """Generate synthetic position state for training."""

    def __init__(self):
        super().__init__()

    def forward(self, batch_size, device):
        """Generate correlated position states for training.

        Returns:
            position_state: (batch, 5) - [is_holding, bars_held_norm, unrealized_pnl_norm,
                                          account_health, loss_streak_frac]
        """
        is_holding = torch.randint(0, 2, (batch_size,), device=device, dtype=torch.float32)

        bars_held_raw = torch.randint(1, 61, (batch_size,), device=device, dtype=torch.float32)
        bars_held_norm = (bars_held_raw / 60.0) * is_holding

        unrealized_pnl_raw = torch.randn(batch_size, device=device) * 0.5 - 0.1
        unrealized_pnl_norm = torch.tanh(unrealized_pnl_raw) * is_holding

        account_health = torch.empty(batch_size, device=device).uniform_(0.1, 1.5)
        loss_streak_base = torch.empty(batch_size, device=device).uniform_(0.0, 1.0)
        health_penalty = torch.clamp(1.0 - account_health, min=0.0)
        loss_streak_frac = torch.clamp(loss_streak_base + health_penalty * 0.5, max=1.0)

        return torch.stack([is_holding, bars_held_norm, unrealized_pnl_norm,
                           account_health, loss_streak_frac], dim=1)


class FeatureNoiseInjection(nn.Module):
    """Inject controlled noise into features during training for temporal robustness."""
    
    def __init__(self, noise_std=0.05):
        super().__init__()
        self.noise_std = noise_std
        
        # Define which feature groups should get noise injection
        # Focus on market microstructure features that may be regime-specific
        self.noise_groups = ['returns', 'volume', 'vol', 'vwap', 'levels', 'micro', 'greeks']
        
    def forward(self, features):
        """
        Args:
            features: (batch, lookback, num_features)
        
        Returns:
            features with noise injected during training
        """
        if not self.training or self.noise_std <= 0:
            return features
            
        noisy_features = features.clone()
        
        for group_name in self.noise_groups:
            if group_name in FEATURE_GROUPS:
                start_idx, end_idx = FEATURE_GROUPS[group_name]
                
                # Generate noise scaled to the feature's natural variance
                feature_slice = features[:, :, start_idx:end_idx]
                feature_std = torch.std(feature_slice, dim=(0, 1), keepdim=True).clamp(min=1e-6)
                
                # Inject multiplicative noise (preserves sign and scale)
                noise_factor = 1.0 + torch.randn_like(feature_slice) * self.noise_std
                noisy_features[:, :, start_idx:end_idx] = feature_slice * noise_factor
                
        return noisy_features


class BalancedStrikeGate(nn.Module):
    """Balanced strike biasing - softer bias toward OTM when account stressed."""
    
    def __init__(self, d_model):
        super().__init__()
        self.health_proj = nn.Linear(1, d_model // 8)
        self.gate_mod = nn.Sequential(
            nn.Linear(d_model + d_model // 8, d_model // 4),
            nn.Tanh(),
            nn.Linear(d_model // 4, 1),
            nn.Sigmoid()
        )
        
    def forward(self, gate_input, dir_input, account_health):
        """
        Args:
            gate_input: (batch, d_model) - input to gate head
            dir_input: (batch, d_model) - input to direction head  
            account_health: (batch, 1) - account health fraction
        
        Returns:
            modified_gate_input: (batch, d_model) - gate input with health modulation
            modified_dir_input: (batch, d_model) - direction input with strike bias
        """
        health_emb = F.relu(self.health_proj(account_health))
        combined = torch.cat([gate_input, health_emb], dim=-1)
        
        # Health gate: closer to 0 when account is stressed (health < 0.6)
        health_gate = self.gate_mod(combined)
        
        # When health is low, reduce the trading signal strength
        health_multiplier = 0.5 + 0.5 * health_gate  # Range: [0.5, 1.0]
        
        return gate_input * health_multiplier, dir_input


class TradingModel(nn.Module):
    """Simplified two-head model for SPX 0DTE options.

    v4 design principles:
    - Simple linear projection instead of FeatureGroupGating (less capacity to memorize)
    - 3 transformer layers instead of 6
    - Dropout 0.15 throughout
    - Position state injection for gate head (kept — this is real signal)
    - No DynamicStopModule, no QualityGate, no Greeks-adaptive anything
    - NEW: Balanced strike gating for capital preservation
    - NEW: Feature noise injection for temporal robustness

    Input:  (batch, lookback, NUM_FEATURES)
    Output: (gate_logits, dir_logits)
        gate_logits: (batch, 2) — [NO_TRADE, TRADE]
        dir_logits:  (batch, 6) — [CALL_ATM, CALL_OTM5, CALL_OTM10,
                                     PUT_ATM, PUT_OTM5, PUT_OTM10]
    """

    def __init__(self, num_features=NUM_FEATURES, lookback=LOOKBACK,
                 d_model=D_MODEL, n_heads=N_HEADS, n_layers=DEPTH,
                 ff_mult=FF_MULT, dropout=DROPOUT):
        super().__init__()
        self.lookback = lookback
        self.d_model = d_model

        # Feature noise injection for temporal robustness
        self.feature_noise = FeatureNoiseInjection(noise_std=FEATURE_NOISE_STD)

        # Simple linear projection — no feature group gating
        self.input_proj = nn.Sequential(
            nn.Linear(num_features, d_model),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.input_norm = nn.LayerNorm(d_model)
        self.pos_embed = nn.Parameter(torch.randn(1, lookback, d_model) * 0.02)

        layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=n_heads,
            dim_feedforward=d_model * ff_mult, dropout=dropout,
            batch_first=True, activation='gelu', norm_first=True,
        )
        self.transformer = nn.TransformerEncoder(layer, num_layers=n_layers)
        mask = nn.Transformer.generate_square_subsequent_mask(lookback)
        self.register_buffer('causal_mask', mask)

        # Position state injection for gate head
        self.position_proj = nn.Linear(5, d_model // 4)
        self.position_gate_proj = nn.Linear(d_model + d_model // 4, d_model)
        self.position_state_gen = PositionStateGenerator()

        # Balanced strike gate
        self.balanced_gate = BalancedStrikeGate(d_model)

        # Gate head: "should I trade?" → [NO_TRADE, TRADE]
        self.gate_head = nn.Sequential(
            nn.LayerNorm(d_model),
            nn.Linear(d_model, d_model // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(d_model // 2, 2),
        )

        # Direction head: "which strike?" → 6-class
        self.dir_head = nn.Sequential(
            nn.LayerNorm(d_model),
            nn.Linear(d_model, d_model // 2),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(d_model // 2, 6),
        )

        # Apply MAXIMUM conservative biases for ultra-selectivity
        with torch.no_grad():
            self.gate_head[-1].bias[0] += 1.2   # NO_TRADE (increased from 1.0)
            self.gate_head[-1].bias[1] -= 1.2   # TRADE (increased from -1.0)
            
            # Apply exact biases from experiment #3
            self.dir_head[-1].bias[0] -= 0.25   # CALL_ATM
            self.dir_head[-1].bias[3] -= 0.25   # PUT_ATM
            self.dir_head[-1].bias[4] += 0.35   # PUT_OTM5

    def forward(self, x, position_state=None):
        batch_size = x.shape[0]
        device = x.device

        # Generate synthetic position state for training if not provided
        if position_state is None and self.training:
            position_state = self.position_state_gen(batch_size, device)

        # Apply feature noise injection for temporal robustness
        x = self.feature_noise(x)

        x_proj = self.input_proj(x)
        x_proj = self.input_norm(x_proj)
        x_proj = x_proj + self.pos_embed[:, :x_proj.size(1), :]
        x_proj = self.transformer(x_proj, mask=self.causal_mask[:x_proj.size(1), :x_proj.size(1)],
                                  is_causal=True)
        last = x_proj[:, -1, :]  # (batch, d_model)

        # Inject position state into gate head with conservative scaling
        if position_state is not None:
            pos_emb = torch.relu(self.position_proj(position_state))
            # Scale down position influence to prevent overriding conservative gate bias
            gate_input = self.position_gate_proj(torch.cat([last, pos_emb * 0.7], dim=-1))
            
            # Apply balanced strike gating using account health
            account_health = position_state[:, 3:4]  # Extract account_health (dim 3)
            gate_input, dir_input = self.balanced_gate(gate_input, last, account_health)
            
            # Apply softer dynamic strike bias when account health < 0.7 (higher threshold)
            health_val = account_health.squeeze(-1)  # (batch,)
            stressed_mask = health_val < 0.7
            
        else:
            gate_input = last
            dir_input = last
            stressed_mask = torch.zeros(batch_size, dtype=torch.bool, device=device)

        gate_logits = self.gate_head(gate_input)
        dir_logits = self.dir_head(dir_input)
        
        # Apply softer strike bias when account is stressed - gentle nudge toward cheaper options
        if stressed_mask.any():
            bias_adjustment = torch.zeros_like(dir_logits)
            # Softer bias: less aggressive penalties and bonuses to maintain ATM viability
            bias_adjustment[stressed_mask, 0] -= 0.15  # CALL_ATM - mild penalty
            bias_adjustment[stressed_mask, 3] -= 0.15  # PUT_ATM - mild penalty
            bias_adjustment[stressed_mask, 1] += 0.10  # CALL_OTM5 - small bonus
            bias_adjustment[stressed_mask, 2] += 0.15  # CALL_OTM10 - modest bonus
            bias_adjustment[stressed_mask, 4] += 0.10  # PUT_OTM5 - small bonus
            bias_adjustment[stressed_mask, 5] += 0.15  # PUT_OTM10 - modest bonus
            
            dir_logits = dir_logits + bias_adjustment

        return gate_logits, dir_logits

def compute_vix_regime_weights(vix_features, target_low_vix_weight=3.0):
    """Compute sample weights to balance VIX regimes in training.
    
    Args:
        vix_features: (batch,) - normalized VIX values
        target_low_vix_weight: upweight factor for low-VIX samples
    
    Returns:
        regime_weights: (batch,) - sample weights for VIX regime balancing
    """
    # VIX thresholds (normalized): <-0.5 = low, -0.5 to 0.5 = normal, >0.5 = high
    low_vix_mask = vix_features < -0.5
    high_vix_mask = vix_features > 0.5
    normal_vix_mask = ~(low_vix_mask | high_vix_mask)
    
    weights = torch.ones_like(vix_features)
    
    # Upweight low-VIX samples (underrepresented but critical for validation)
    weights[low_vix_mask] = target_low_vix_weight
    
    # Slightly downweight high-VIX samples (overrepresented in training)
    weights[high_vix_mask] = 0.7
    
    # Normal VIX gets baseline weight
    weights[normal_vix_mask] = 1.0
    
    return weights


def vix_consistency_loss(gate_logits, dir_logits, vix_features):
    """Regularizer to encourage consistent behavior across VIX regimes.
    
    Penalizes the model for having dramatically different gate/direction preferences
    between low and high VIX periods.
    """
    if len(gate_logits) < 10:  # Need sufficient samples
        return torch.tensor(0.0, device=gate_logits.device)
    
    low_vix_mask = vix_features < -0.5
    high_vix_mask = vix_features > 0.5
    
    if low_vix_mask.sum() < 2 or high_vix_mask.sum() < 2:
        return torch.tensor(0.0, device=gate_logits.device)
    
    # Gate consistency: similar trading frequency across regimes
    gate_probs = F.softmax(gate_logits, dim=-1)
    low_vix_trade_rate = gate_probs[low_vix_mask, 1].mean()
    high_vix_trade_rate = gate_probs[high_vix_mask, 1].mean()
    
    gate_consistency = torch.abs(low_vix_trade_rate - high_vix_trade_rate)
    
    # Direction consistency: similar strike preferences across regimes
    dir_probs = F.softmax(dir_logits, dim=-1)
    low_vix_dir_dist = dir_probs[low_vix_mask].mean(dim=0)
    high_vix_dir_dist = dir_probs[high_vix_mask].mean(dim=0)
    
    dir_consistency = F.kl_div(
        F.log_softmax(low_vix_dir_dist.unsqueeze(0), dim=-1),
        F.softmax(high_vix_dir_dist.unsqueeze(0), dim=-1),
        reduction='batchmean'
    )
    
    return gate_consistency + dir_consistency


def temporal_chunk_consistency_loss(gate_logits, dir_logits, time_features):
    """Regularizer to encourage consistent behavior across temporal chunks.
    
    Splits training data into early/mid/late chunks based on minutes_to_close
    and penalizes dramatic differences in trading behavior between chunks.
    This addresses the Chunk 1 weakness (PF=0.22) by preventing temporal overfitting.
    """
    if len(gate_logits) < 15:  # Need sufficient samples for 3 chunks
        return torch.tensor(0.0, device=gate_logits.device)
    
    # Split by time of day (minutes to close normalized 0-1, where 1=market open, 0=close)
    # Early session: > 0.67 (first ~2 hours)
    # Mid session: 0.33-0.67 (middle ~2.5 hours) 
    # Late session: < 0.33 (final ~2 hours)
    early_mask = time_features > 0.67
    mid_mask = (time_features >= 0.33) & (time_features <= 0.67)
    late_mask = time_features < 0.33
    
    # Need at least 2 samples in each chunk
    if early_mask.sum() < 2 or mid_mask.sum() < 2 or late_mask.sum() < 2:
        return torch.tensor(0.0, device=gate_logits.device)
    
    gate_probs = F.softmax(gate_logits, dim=-1)
    dir_probs = F.softmax(dir_logits, dim=-1)
    
    # Gate consistency across temporal chunks
    early_trade_rate = gate_probs[early_mask, 1].mean()
    mid_trade_rate = gate_probs[mid_mask, 1].mean()
    late_trade_rate = gate_probs[late_mask, 1].mean()
    
    gate_consistency = (torch.abs(early_trade_rate - mid_trade_rate) + 
                       torch.abs(mid_trade_rate - late_trade_rate) + 
                       torch.abs(early_trade_rate - late_trade_rate)) / 3.0
    
    # Direction consistency across temporal chunks
    early_dir_dist = dir_probs[early_mask].mean(dim=0)
    mid_dir_dist = dir_probs[mid_mask].mean(dim=0)
    late_dir_dist = dir_probs[late_mask].mean(dim=0)
    
    # KL divergence between distributions
    dir_consistency = (
        F.kl_div(F.log_softmax(early_dir_dist.unsqueeze(0), dim=-1), 
                 F.softmax(mid_dir_dist.unsqueeze(0), dim=-1), reduction='batchmean') +
        F.kl_div(F.log_softmax(mid_dir_dist.unsqueeze(0), dim=-1),
                 F.softmax(late_dir_dist.unsqueeze(0), dim=-1), reduction='batchmean') +
        F.kl_div(F.log_softmax(early_dir_dist.unsqueeze(0), dim=-1),
                 F.softmax(late_dir_dist.unsqueeze(0), dim=-1), reduction='batchmean')
    ) / 3.0
    
    return gate_consistency + dir_consistency


def direction_side_loss(dir_logits, call_pnl, put_pnl):
    """Penalize call/put confusion — the model should learn the correct SIDE first.

    Direction collapse (50%) means the model can't distinguish bullish from bearish setups.
    This loss trains a binary call-vs-put signal before worrying about strike selection.
    """
    if len(dir_logits) < 5:
        return torch.tensor(0.0, device=dir_logits.device)

    # Aggregate call-side vs put-side probabilities
    dir_probs = F.softmax(dir_logits, dim=-1)
    call_prob = dir_probs[:, :3].sum(dim=-1)  # CALL_ATM + CALL_OTM5 + CALL_OTM10
    put_prob = dir_probs[:, 3:].sum(dim=-1)   # PUT_ATM + PUT_OTM5 + PUT_OTM10

    # Best side based on P&L: which side was actually profitable?
    call_better = call_pnl > put_pnl

    # Binary cross-entropy: reward model for picking the correct side
    side_pred = call_prob  # probability model assigns to calls
    side_target = call_better.float()

    # Only train on bars where there's a meaningful difference between sides
    side_gap = torch.abs(call_pnl - put_pnl)
    meaningful = side_gap > 0.05  # At least 5% P&L difference between sides

    if meaningful.sum() < 2:
        return torch.tensor(0.0, device=dir_logits.device)

    side_loss = F.binary_cross_entropy(
        side_pred[meaningful].clamp(1e-6, 1 - 1e-6),
        side_target[meaningful],
        reduction='mean'
    )

    return side_loss


def strike_consistency_loss(dir_logits, time_features, vix_features):
    """NEW: Regularize strike selection to be consistent within similar market conditions.
    
    Penalizes erratic switching between ATM and OTM strikes when market conditions
    (time of day + VIX regime) are similar. Encourages the model to develop coherent
    strike selection patterns rather than random choices.
    """
    if len(dir_logits) < 20:  # Need sufficient samples
        return torch.tensor(0.0, device=dir_logits.device)
    
    # Create similarity groups based on time + VIX
    # Time buckets: early (>0.7), mid (0.3-0.7), late (<0.3)
    # VIX buckets: low (<-0.3), normal (-0.3 to 0.3), high (>0.3)
    
    time_early = time_features > 0.7
    time_mid = (time_features >= 0.3) & (time_features <= 0.7)
    time_late = time_features < 0.3
    
    vix_low = vix_features < -0.3
    vix_normal = (vix_features >= -0.3) & (vix_features <= 0.3)
    vix_high = vix_features > 0.3
    
    dir_probs = F.softmax(dir_logits, dim=-1)
    
    # Aggregate ATM vs OTM preferences
    atm_prob = dir_probs[:, 0] + dir_probs[:, 3]  # CALL_ATM + PUT_ATM
    otm_prob = 1.0 - atm_prob  # All OTM strikes
    
    consistency_loss = 0.0
    num_groups = 0
    
    # For each time-VIX combination, measure consistency of ATM vs OTM preference
    for time_mask in [time_early, time_mid, time_late]:
        for vix_mask in [vix_low, vix_normal, vix_high]:
            group_mask = time_mask & vix_mask
            if group_mask.sum() < 3:  # Need at least 3 samples
                continue
                
            group_atm_probs = atm_prob[group_mask]
            # Penalize high variance in ATM vs OTM preference within similar conditions
            atm_variance = torch.var(group_atm_probs)
            consistency_loss += atm_variance
            num_groups += 1
    
    if num_groups == 0:
        return torch.tensor(0.0, device=dir_logits.device)
    
    return consistency_loss / num_groups


def stop_prevention_loss(gate_logits, call_stopped_pnl, put_stopped_pnl, call_raw_pnl, put_raw_pnl):
    """Penalize the model for making trades that would hit dynamic stops.
    
    Compares stopped P&L vs raw P&L to identify trades killed by stops.
    Penalizes the gate for predicting TRADE on such samples.
    """
    if call_stopped_pnl is None or put_stopped_pnl is None or call_raw_pnl is None or put_raw_pnl is None:
        return torch.tensor(0.0, device=gate_logits.device)
    
    # Identify trades that hit stops: large negative difference between stopped and raw P&L
    call_stopped_by_stop = (call_stopped_pnl - call_raw_pnl) < -0.3  # Lost >30% due to stop
    put_stopped_by_stop = (put_stopped_pnl - put_raw_pnl) < -0.3
    
    # Any option hitting stop means this bar would have been a bad trade
    hit_stop = call_stopped_by_stop | put_stopped_by_stop
    
    if hit_stop.sum() < 2:
        return torch.tensor(0.0, device=gate_logits.device)
    
    # For bars that would hit stops, penalize gate for predicting TRADE
    stop_gate_logits = gate_logits[hit_stop]
    stop_targets = torch.zeros(hit_stop.sum().item(), dtype=torch.long, device=gate_logits.device)  # NO_TRADE
    
    stop_loss = F.cross_entropy(stop_gate_logits, stop_targets, reduction='mean')
    
    return stop_loss


def _get_stopped_or_raw(stopped_pnl, raw_pnl):
    """Use stopped P&L if available, otherwise fall back to raw P&L."""
    if stopped_pnl is not None:
        return stopped_pnl
    else:
        return raw_pnl


def sniper_loss(gate_logits, dir_logits, call_pnl, put_pnl, time_features, features,
                exit_call_labels=None, exit_put_labels=None,
                otm5_call_pnl=None, otm5_put_pnl=None,
                otm10_call_pnl=None, otm10_put_pnl=None,
                supervision_weight=None, actionable_mask=None, risk_state_mask=None,
                call_stopped_pnl=None, put_stopped_pnl=None,
                otm5_call_stopped_pnl=None, otm5_put_stopped_pnl=None,
                otm10_call_stopped_pnl=None, otm10_put_stopped_pnl=None):
    """Two-head loss with FULL STOPPED P&L CONSISTENCY + direction-side loss + strike consistency.

    CRITICAL CHANGE: Use ONLY stopped P&L for all training targets to align with evaluation reality.
    This eliminates conflicting signals between "hold through drawdowns" (raw P&L) and 
    "trades get stopped out" (evaluation behavior).
    """
    B = gate_logits.shape[0]
    device = gate_logits.device

    # Use stopped P&L consistently throughout
    call_pnl_final = _get_stopped_or_raw(call_stopped_pnl, call_pnl)
    put_pnl_final = _get_stopped_or_raw(put_stopped_pnl, put_pnl)
    otm5_call_final = _get_stopped_or_raw(otm5_call_stopped_pnl, otm5_call_pnl)
    otm5_put_final = _get_stopped_or_raw(otm5_put_stopped_pnl, otm5_put_pnl)
    otm10_call_final = _get_stopped_or_raw(otm10_call_stopped_pnl, otm10_call_pnl)
    otm10_put_final = _get_stopped_or_raw(otm10_put_stopped_pnl, otm10_put_pnl)

    # Mask: only compute loss where we have ATM option P&L data
    valid = ~torch.isnan(call_pnl_final) & ~torch.isnan(put_pnl_final)
    if valid.sum() < 2:
        return gate_logits.sum() * 0.0

    g_logits = gate_logits[valid]
    d_logits = dir_logits[valid]
    c_pnl = call_pnl_final[valid]
    p_pnl = put_pnl_final[valid]
    t_feat = time_features[valid]
    feat = features[valid]
    
    # Extract VIX feature for regime balancing
    vix_feat = feat[:, IDX_VIX_NORM] if feat.shape[1] > IDX_VIX_NORM else torch.zeros(feat.shape[0], device=device)
    
    sample_weight = torch.ones_like(c_pnl)

    if supervision_weight is not None:
        sw = torch.nan_to_num(supervision_weight[valid], nan=0.0).clamp(min=0.0)
        if sw.sum() > 0:
            sample_weight = sw
    if actionable_mask is not None:
        act = torch.nan_to_num(actionable_mask[valid], nan=0.0).clamp(0.0, 1.0)
        sample_weight = sample_weight * torch.where(
            act > 0.5, torch.ones_like(act), torch.full_like(act, 0.20))
    if risk_state_mask is not None:
        risk = torch.nan_to_num(risk_state_mask[valid], nan=1.0).clamp(0.0, 1.0)
        sample_weight = sample_weight * torch.where(
            risk > 0.5, torch.ones_like(risk), torch.full_like(risk, 0.50))
    
    # Apply VIX regime balancing
    vix_weights = compute_vix_regime_weights(vix_feat)
    sample_weight = sample_weight * vix_weights
    
    if sample_weight.sum() <= 0:
        sample_weight = torch.ones_like(sample_weight)
    sample_weight = sample_weight / sample_weight.mean().clamp(min=1e-6)

    # Build 6-class P&L array using STOPPED P&L consistently
    all_pnl = torch.stack([
        c_pnl,                             # CALL_ATM (stopped)
        otm5_call_final[valid],           # CALL_OTM5 (stopped)
        otm10_call_final[valid],          # CALL_OTM10 (stopped)
        p_pnl,                             # PUT_ATM (stopped)
        otm5_put_final[valid],            # PUT_OTM5 (stopped)
        otm10_put_final[valid],           # PUT_OTM10 (stopped)
    ], dim=-1)  # (valid, 6)

    # Gate targets: TRADE (1) when ANY option is profitable after stops
    any_profitable = torch.any(torch.nan_to_num(all_pnl, nan=-999.0) > 0, dim=-1)
    gate_targets = any_profitable.long()

    # Asymmetric gate weights: penalize false entries more than missed entries
    n_trade = gate_targets.sum().float().clamp(min=1)
    n_no_trade = (gate_targets == 0).sum().float().clamp(min=1)
    base_balance = (n_no_trade / n_trade).clamp(max=10.0)
    gate_weights = torch.tensor([1.0 / FALSE_ENTRY_PENALTY, base_balance], device=device)

    time_weight = 1.0 + 0.5 * (1.0 - t_feat)
    gate_loss = F.cross_entropy(
        g_logits, gate_targets, weight=gate_weights,
        reduction='none', label_smoothing=GATE_LABEL_SMOOTHING,
    )
    gate_loss = (gate_loss * time_weight * sample_weight).mean()

    # Direction targets: only where gate_target = TRADE
    trade_mask = gate_targets == 1
    if trade_mask.sum() < 2:
        total = GATE_LOSS_WEIGHT * gate_loss
        
        # Add VIX consistency loss
        if VIX_CONSISTENCY_WEIGHT > 0:
            consistency_loss = vix_consistency_loss(g_logits, d_logits, vix_feat)
            total = total + VIX_CONSISTENCY_WEIGHT * consistency_loss
            
        # Add temporal chunk consistency loss
        if TEMPORAL_CHUNK_WEIGHT > 0:
            temporal_loss = temporal_chunk_consistency_loss(g_logits, d_logits, t_feat)
            total = total + TEMPORAL_CHUNK_WEIGHT * temporal_loss
            
        # Add direction-side loss (addresses 50% direction collapse)
        if DIR_SIDE_WEIGHT > 0:
            dir_side = direction_side_loss(d_logits, c_pnl, p_pnl)
            total = total + DIR_SIDE_WEIGHT * dir_side

        # Add strike consistency loss (NEW)
        if STRIKE_CONSISTENCY_WEIGHT > 0:
            strike_consistency = strike_consistency_loss(d_logits, t_feat, vix_feat)
            total = total + STRIKE_CONSISTENCY_WEIGHT * strike_consistency

        # Add stop prevention loss (using raw P&L as reference)
        if STOP_PREVENTION_WEIGHT > 0:
            raw_call = call_pnl[valid] if call_pnl is not None else c_pnl
            raw_put = put_pnl[valid] if put_pnl is not None else p_pnl
            stop_loss = stop_prevention_loss(g_logits, c_pnl, p_pnl, raw_call, raw_put)
            total = total + STOP_PREVENTION_WEIGHT * stop_loss
            
        return total

    d_logits_trade = d_logits[trade_mask]
    t_feat_trade = t_feat[trade_mask]
    trade_pnl = all_pnl[trade_mask]

    trade_pnl_safe = torch.nan_to_num(trade_pnl, nan=-999.0)
    dir_targets = torch.argmax(trade_pnl_safe, dim=-1)

    dir_loss = F.cross_entropy(
        d_logits_trade, dir_targets,
        reduction='none', label_smoothing=DIR_LABEL_SMOOTHING,
    )
    dir_time_weight = 1.0 + 0.5 * (1.0 - t_feat_trade)
    dir_w = sample_weight[trade_mask]
    dir_w = dir_w / dir_w.mean().clamp(min=1e-6)
    dir_loss = (dir_loss * dir_time_weight * dir_w).mean()

    # P&L alignment bonus using STOPPED P&L
    gate_probs = F.softmax(g_logits, dim=-1)
    dir_probs = F.softmax(d_logits, dim=-1)
    trade_prob = gate_probs[:, 1]
    all_pnl_safe = torch.nan_to_num(all_pnl, nan=0.0)
    pnl_signal = trade_prob * (dir_probs * all_pnl_safe).sum(dim=-1)
    pnl_loss = -(pnl_signal * sample_weight).sum() / sample_weight.sum().clamp(min=1e-6)

    # EXIT loss: train gate to predict NO_TRADE on exit signal bars
    exit_loss = torch.tensor(0.0, device=device)
    if exit_call_labels is not None and exit_put_labels is not None:
        ec = exit_call_labels[valid]
        ep = exit_put_labels[valid]
        exit_mask = (ec > 0.5) | (ep > 0.5)
        if exit_mask.sum() > 1:
            exit_g = g_logits[exit_mask]
            exit_targets = torch.zeros(exit_mask.sum().item(), dtype=torch.long, device=device)
            exit_loss_vec = F.cross_entropy(exit_g, exit_targets, reduction='none')
            exit_w = sample_weight[exit_mask]
            exit_loss = (exit_loss_vec * exit_w).sum() / exit_w.sum().clamp(min=1e-6)

    # VIX consistency regularization
    vix_consistency_loss_val = torch.tensor(0.0, device=device)
    if VIX_CONSISTENCY_WEIGHT > 0:
        vix_consistency_loss_val = vix_consistency_loss(g_logits, d_logits, vix_feat)

    # Temporal chunk consistency regularization
    temporal_consistency_loss_val = torch.tensor(0.0, device=device)
    if TEMPORAL_CHUNK_WEIGHT > 0:
        temporal_consistency_loss_val = temporal_chunk_consistency_loss(g_logits, d_logits, t_feat)

    # Direction-side loss (addresses 50% direction collapse)
    dir_side_loss_val = torch.tensor(0.0, device=device)
    if DIR_SIDE_WEIGHT > 0:
        dir_side_loss_val = direction_side_loss(d_logits, c_pnl, p_pnl)

    # Strike consistency loss (NEW)
    strike_consistency_loss_val = torch.tensor(0.0, device=device)
    if STRIKE_CONSISTENCY_WEIGHT > 0:
        strike_consistency_loss_val = strike_consistency_loss(d_logits, t_feat, vix_feat)

    # Stop prevention loss (using raw P&L as reference if available)
    stop_prevention_loss_val = torch.tensor(0.0, device=device)
    if STOP_PREVENTION_WEIGHT > 0:
        raw_call = call_pnl[valid] if call_pnl is not None else c_pnl
        raw_put = put_pnl[valid] if put_pnl is not None else p_pnl
        stop_prevention_loss_val = stop_prevention_loss(g_logits, c_pnl, p_pnl, raw_call, raw_put)

    total = (GATE_LOSS_WEIGHT * gate_loss + DIR_LOSS_WEIGHT * dir_loss
             + PNL_ALIGNMENT_WEIGHT * pnl_loss + EXIT_LOSS_WEIGHT * exit_loss
             + VIX_CONSISTENCY_WEIGHT * vix_consistency_loss_val
             + TEMPORAL_CHUNK_WEIGHT * temporal_consistency_loss_val
             + DIR_SIDE_WEIGHT * dir_side_loss_val
             + STRIKE_CONSISTENCY_WEIGHT * strike_consistency_loss_val
             + STOP_PREVENTION_WEIGHT * stop_prevention_loss_val)
    return total

def get_lr_mult(progress):
    if progress < WARMUP_RATIO:
        return progress / max(WARMUP_RATIO, 1e-8)
    if progress < 1.0 - COOLDOWN_RATIO:
        return 1.0
    else:
        t = (1.0 - progress) / max(COOLDOWN_RATIO, 1e-8)
        return 0.5 * (1.0 + math.cos(math.pi * (1.0 - t)))
